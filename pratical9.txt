```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Personal To-Do List</title>

    <!-- Tailwind CSS -->
    <script src="https://cdn.tailwindcss.com"></script>
</head>

<body class="bg-gray-100 min-h-screen flex items-center justify-center p-5">

    <!-- Main Container -->
    <div class="bg-white w-full max-w-lg rounded-2xl shadow-xl p-6">

        <!-- Heading -->
        <h1 class="text-3xl font-bold text-center text-blue-600 mb-6">
            📝 Personal To-Do List
        </h1>

        <!-- Input Section -->
        <div class="flex gap-2 mb-6">

            <input
                type="text"
                id="taskInput"
                placeholder="Enter a task..."
                class="flex-1 border border-gray-300 rounded-lg px-4 py-3
                       focus:outline-none focus:ring-2 focus:ring-blue-500"
            >

            <button
                onclick="addTask()"
                class="bg-blue-600 text-white px-5 py-3 rounded-lg
                       hover:bg-blue-700 transition font-semibold"
            >
                Add
            </button>

        </div>

        <!-- Task List -->
        <ul id="taskList" class="space-y-3"></ul>

        <!-- Empty Message -->
        <p id="emptyMessage" class="text-center text-gray-400 mt-5">
            No tasks added yet.
        </p>

    </div>


    <script>

        // Add Task
        function addTask() {

            const input = document.getElementById("taskInput");
            const taskText = input.value.trim();

            // Check empty input
            if (taskText === "") {
                alert("Please enter a task!");
                return;
            }

            // Create list item
            const li = document.createElement("li");

            li.className =
                "flex items-center justify-between bg-gray-50 border " +
                "border-gray-200 rounded-lg p-3";

            // Task text
            const span = document.createElement("span");

            span.textContent = taskText;

            span.className =
                "text-gray-700 cursor-pointer flex-1";

            // Mark task completed
            span.onclick = function () {
                span.classList.toggle("line-through");
                span.classList.toggle("text-gray-400");
            };


            // Delete button
            const deleteButton = document.createElement("button");

            deleteButton.textContent = "Delete";

            deleteButton.className =
                "bg-red-500 text-white px-3 py-2 rounded-lg " +
                "hover:bg-red-600 transition ml-3";

            // Delete task
            deleteButton.onclick = function () {
                li.remove();
                updateEmptyMessage();
            };


            // Add elements
            li.appendChild(span);
            li.appendChild(deleteButton);

            document.getElementById("taskList").appendChild(li);

            // Clear input
            input.value = "";

            updateEmptyMessage();
        }


        // Show / hide empty message
        function updateEmptyMessage() {

            const taskList = document.getElementById("taskList");
            const emptyMessage = document.getElementById("emptyMessage");

            if (taskList.children.length === 0) {
                emptyMessage.style.display = "block";
            } else {
                emptyMessage.style.display = "none";
            }
        }


        // Press Enter to add task
        document.getElementById("taskInput").addEventListener("keypress", function(event) {

            if (event.key === "Enter") {
                addTask();
            }

        });

    </script>

</body>
</html>
```
