    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Personal Expense Tracker</title>

    <!-- Tailwind CSS CDN -->
    <script src="https://cdn.tailwindcss.com"></script>
</head>

<body class="bg-gray-100 min-h-screen">

    <!-- Header -->
    <header class="bg-blue-600 text-white py-6 shadow-lg">
        <div class="max-w-6xl mx-auto px-4">
            <h1 class="text-3xl md:text-4xl font-bold text-center">
                💰 Personal Expense Tracker
            </h1>

            <p class="text-center mt-2 text-blue-100">
                Track your daily expenses easily
            </p>
        </div>
    </header>


    <!-- Main Container -->
    <main class="max-w-6xl mx-auto px-4 py-8">

        <!-- Summary Cards -->
        <div class="grid grid-cols-1 sm:grid-cols-3 gap-5 mb-8">

            <!-- Total Expenses -->
            <div class="bg-white rounded-xl shadow p-6">
                <h2 class="text-gray-500 text-sm font-semibold">
                    TOTAL EXPENSES
                </h2>

                <p id="totalExpense"
                   class="text-3xl font-bold text-red-500 mt-2">
                    ₹0
                </p>
            </div>


            <!-- Number of Expenses -->

                    TOTAL TRANSACTIONS
                </h2>

                <p id="totalTransactions"
                   class="text-3xl font-bold text-blue-600 mt-2">
                    0
                </p>
            </div>


            <!-- Average Expense -->
            <div class="bg-white rounded-xl shadow p-6">
                <h2 class="text-gray-500 text-sm font-semibold">
                    AVERAGE EXPENSE
                </h2>

                <p id="averageExpense"
                   class="text-3xl font-bold text-green-600 mt-2">
                    ₹0
                </p>
            </div>

        </div>


        <!-- Add Expense Form -->
        <div class="bg-white rounded-xl shadow p-6 mb-8">

            <h2 class="text-2xl font-bold text-gray-800 mb-5">
                ➕ Add New Expense
            </h2>

            <div class="grid grid-cols-1 md:grid-cols-4 gap-4">

                <!-- Description -->
                <input
                    type="text"
                    id="expenseName"
                    placeholder="Expense name"
                    class="border border-gray-300 rounded-lg px-4 py-3

                <input

                    id="expenseAmount"

                    <option value="">Select Category</option>

    <!-- JavaScript -->
    <script>

        // Store expenses
        let expenses = [];


        // Add Expense Function
        function addExpense() {

            const name =
                document.getElementById("expenseName").value.trim();

            const amount =
                parseFloat(document.getElementById("expenseAmount").value);

            const category =
                document.getElementById("expenseCategory").value;


            // Validation
            if (name === "" || isNaN(amount) || amount <= 0 || category === "") {

                alert("Please enter valid expense details.");

                return;
            }


            // Create expense object
            const expense = {

                id: Date.now(),

                name: name,

                amount: amount,

                category: category

            };


            // Add expense to array
            expenses.push(expense);


            // Clear input fields
            document.getElementById("expenseName").value = "";

            document.getElementById("expenseAmount").value = "";

            document.getElementById("expenseCategory").value = "";


            // Display expenses
            displayExpenses();

        }


        // Display Expenses
        function displayExpenses() {

            const list =
                document.getElementById("expenseList");

            const emptyMessage =
                document.getElementById("emptyMessage");


            // Clear old list
            list.innerHTML = "";


            // Empty list
            if (expenses.length === 0) {

                emptyMessage.style.display = "block";

            } else {

                emptyMessage.style.display = "none";


                // Display every expense
                expenses.forEach(function(expense) {

                    const row =
                        document.createElement("tr");

                    row.className =
                        "border-t hover:bg-gray-50";


                    row.innerHTML = `

                        <td class="px-6 py-4 font-medium">
                            ${expense.name}
                        </td>

                        <td class="px-6 py-4">
                            <span class="bg-blue-100 text-blue-700
                                         px-3 py-1 rounded-full text-sm">
                                ${expense.category}
                            </span>
                        </td>

                        <td class="px-6 py-4 font-bold text-red-500">
                            ₹${expense.amount.toFixed(2)}
                        </td>

                        <td class="px-6 py-4">

                            <button
                                onclick="deleteExpense(${expense.id})"
                                class="bg-red-500 text-white px-3 py-2
                                       rounded-lg hover:bg-red-600">
                                Delete
                            </button>

                        </td>
                    `;


                    list.appendChild(row);

                });

            }


            // Update summary
            updateSummary();

        }


        // Delete Expense
        function deleteExpense(id) {

            expenses =
                expenses.filter(function(expense) {

                    return expense.id !== id;

                });


            displayExpenses();

        }


        // Update Summary
        function updateSummary() {

            let total = 0;


            expenses.forEach(function(expense) {

                total += expense.amount;

            });


            document.getElementById("totalExpense").textContent =
                "₹" + total.toFixed(2);


            document.getElementById("totalTransactions").textContent =
                expenses.length;


            let average = 0;


            if (expenses.length > 0) {

                average = total / expenses.length;

            }


            document.getElementById("averageExpense").textContent =
                "₹" + average.toFixed(2);

        }

    </script>

</body>
